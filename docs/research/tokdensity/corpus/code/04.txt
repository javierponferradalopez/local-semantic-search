import { describe, it, expect, beforeEach } from 'vitest';
import Database from 'better-sqlite3';
import { readFileSync } from 'node:fs';
import { SqliteChunkRepository, splitIntoChunks, type Chunk } from '../src/chunks.js';

const SCHEMA = readFileSync(new URL('../src/schema.sql', import.meta.url), 'utf8');

describe('SqliteChunkRepository', () => {
  let db: Database.Database;
  let repo: SqliteChunkRepository;

  beforeEach(() => {
    db = new Database(':memory:');
    db.exec(SCHEMA);
    db.prepare(
      `INSERT INTO documents (id, source_path, mime_type, size_bytes, sha256)
       VALUES ('d1', '/tmp/a.md', 'text/markdown', 10, 'abc')`,
    ).run();
    repo = new SqliteChunkRepository(db);
  });

  it('stores chunks and reads them back in order', () => {
    const chunks: Chunk[] = [
      { id: 'c2', documentId: 'd1', ordinal: 1, start: 5, end: 10, text: 'world' },
      { id: 'c1', documentId: 'd1', ordinal: 0, start: 0, end: 5, text: 'hello' },
    ];
    repo.saveMany(chunks);
    const found = repo.findByDocument('d1');
    expect(found.map((c) => c.id)).toEqual(['c1', 'c2']);
  });

  it('rejects a chunk whose end is not after its start', () => {
    expect(() =>
      repo.saveMany([{ id: 'c1', documentId: 'd1', ordinal: 0, start: 5, end: 5, text: '' }]),
    ).toThrow(/CHECK constraint failed/);
  });

  it('deletes all chunks of a document and reports the count', () => {
    repo.saveMany(splitIntoChunks('d1', 'a'.repeat(2500), 1000));
    expect(repo.deleteByDocument('d1')).toBe(3);
    expect(repo.findByDocument('d1')).toHaveLength(0);
  });
});

describe('splitIntoChunks', () => {
  it('returns one chunk for text under the cap', () => {
    const out = splitIntoChunks('d', 'short text', 1000);
    expect(out).toHaveLength(1);
    expect(out[0]).toMatchObject({ start: 0, end: 10, ordinal: 0 });
  });

  it('prefers a paragraph break in the second half of the window', () => {
    const text = `${'x'.repeat(700)}\n\n${'y'.repeat(700)}`;
    const out = splitIntoChunks('d', text, 1000);
    expect(out[0].end).toBe(700);
    expect(out[1].text.startsWith('\n\nyyy')).toBe(true);
  });

  it('covers the whole text without gaps or overlaps', () => {
    const text = Array.from({ length: 40 }, (_, i) => `Paragraph ${i}.`).join('\n\n');
    const out = splitIntoChunks('d', text, 120);
    for (let i = 1; i < out.length; i++) expect(out[i].start).toBe(out[i - 1].end);
    expect(out.map((c) => c.text).join('')).toBe(text);
  });
});
